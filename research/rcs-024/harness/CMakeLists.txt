cmake_minimum_required(VERSION 3.20)
project(rcs024_occt_differential LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
set(RCS024_OCCT_ROOT "" CACHE PATH "Exact OCCT installation root")
set(RCS024_EXPECTED_VERSION "" CACHE STRING "Expected OCC_VERSION_COMPLETE")
set(RCS024_EXPECTED_COMMIT "" CACHE STRING "Exact upstream commit")
set(RCS024_LABEL "" CACHE STRING "Evidence label")
if(NOT RCS024_OCCT_ROOT OR NOT RCS024_EXPECTED_VERSION OR NOT RCS024_EXPECTED_COMMIT OR NOT RCS024_LABEL)
  message(FATAL_ERROR "Set RCS024_OCCT_ROOT, RCS024_EXPECTED_VERSION, RCS024_EXPECTED_COMMIT and RCS024_LABEL")
endif()
find_path(OCCT_INCLUDE NAMES Standard.hxx PATHS "${RCS024_OCCT_ROOT}/include/opencascade" NO_DEFAULT_PATH REQUIRED)
set(TOOLKITS TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKShHealing TKDE TKXSBase TKDESTEP)
set(OCCT_LIBS "")
foreach(t IN LISTS TOOLKITS)
  find_library(LIB_${t} NAMES ${t} PATHS "${RCS024_OCCT_ROOT}/lib" "${RCS024_OCCT_ROOT}/lib64" NO_DEFAULT_PATH REQUIRED)
  list(APPEND OCCT_LIBS "${LIB_${t}}")
endforeach()
add_executable(rcs024_diff_worker diff_worker.cpp)
add_executable(rcs024_brepgraph_probe brepgraph_probe.cpp)
foreach(target rcs024_diff_worker rcs024_brepgraph_probe)
  target_include_directories(${target} PRIVATE "${OCCT_INCLUDE}")
  target_link_libraries(${target} PRIVATE ${OCCT_LIBS} Threads::Threads)
  target_compile_definitions(${target} PRIVATE
    RCS024_EXPECTED_VERSION="${RCS024_EXPECTED_VERSION}"
    RCS024_EXPECTED_COMMIT="${RCS024_EXPECTED_COMMIT}"
    RCS024_LABEL="${RCS024_LABEL}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
