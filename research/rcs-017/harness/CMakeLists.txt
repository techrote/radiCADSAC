cmake_minimum_required(VERSION 3.20)
project(rcs017_concurrency_probe LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

set(RCS006_OCCT_ROOT "" CACHE PATH "Pinned OCCT installation root")
if(NOT RCS006_OCCT_ROOT AND DEFINED ENV{RCS006_OCCT_PREFIX})
  set(RCS006_OCCT_ROOT "$ENV{RCS006_OCCT_PREFIX}")
endif()
if(NOT RCS006_OCCT_ROOT AND CMAKE_PREFIX_PATH)
  list(GET CMAKE_PREFIX_PATH 0 RCS006_OCCT_ROOT)
endif()
if(NOT RCS006_OCCT_ROOT)
  message(FATAL_ERROR "Set RCS006_OCCT_ROOT or RCS006_OCCT_PREFIX")
endif()

find_path(OCCT_INCLUDE_DIR NAMES Standard.hxx PATHS "${RCS006_OCCT_ROOT}/include/opencascade" NO_DEFAULT_PATH REQUIRED)
set(OCCT_TOOLKITS TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKShHealing TKDE TKXSBase TKDESTEP)
set(OCCT_LIBRARIES "")
foreach(toolkit IN LISTS OCCT_TOOLKITS)
  find_library(OCCT_${toolkit}_LIBRARY NAMES ${toolkit} PATHS "${RCS006_OCCT_ROOT}/lib" "${RCS006_OCCT_ROOT}/lib64" NO_DEFAULT_PATH REQUIRED)
  list(APPEND OCCT_LIBRARIES "${OCCT_${toolkit}_LIBRARY")
endforeach()

add_executable(rcs017_concurrency_probe concurrency_probe.cpp)
target_include_directories(rcs017_concurrency_probe PRIVATE "${OCCT_INCLUDE_DIR}")
target_link_libraries(rcs017_concurrency_probe PRIVATE ${OCCT_LIBRARIES} Threads::Threads)
if(MSVC)
  target_compile_options(rcs017_concurrency_probe PRIVATE /W4 /permissive-)
else()
  target_compile_options(rcs017_concurrency_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
